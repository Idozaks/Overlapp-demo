    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const observerOptions = {
                root: null,
                rootMargin: '0px',
                threshold: 0.1 // Adjust threshold for video if needed (e.g., 0.2)
            };

            const observerCallback = (entries, observer) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add('is-visible');
                        observer.unobserve(entry.target);
                    }
                });
            };

            const observer = new IntersectionObserver(observerCallback, observerOptions);

            // Add .video-wrapper to the elements observed
            const elementsToAnimate = document.querySelectorAll('.animate-on-scroll, section h2, section p.section-intro, .feature-item, #privacy .privacy-icon, .form-container, .video-wrapper');
            elementsToAnimate.forEach(el => {
                observer.observe(el);
            });
        });
    </script>
